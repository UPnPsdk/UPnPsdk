21// Constructor for getting an address information from only a netaddress.
22// ----------------------------------------------------------------------
23CAddrinfo::CAddrinfo(const int a_flags, const int a_socktype) noexcept {
24 // I cannot use the initialization list of the constructor because the
25 // member order in the structure addrinfo is different on Linux, MacOS and
26 // win32. I have to use the member names to initialize them, what's not
27 // possible for structures in the constructors initialization list.
28 m_hints.ai_socktype = a_socktype;
29 // Due to specification IPv4 addresses are always mapped to IPv6.
30 m_hints.ai_family = AF_INET6;
31 m_hints.ai_flags = AI_V4MAPPED;
32 m_hints.ai_flags = AI_V4MAPPED | a_flags;
33 // In contrast to other platforms ::getaddrinfo() on win32 does not create
34 // AI_V4MAPPED addresses with AI_NUMERICHOST set. UPnPsdk only uses IPv6
35 // addresses. All IPv4 addresses are mapped to IPv6. There is only one
36 // combination with AF_INET6 and no AI_NUMERICHOST where win32 do
37 // AI_V4MAPPED. All others fail. To be portable, UPnPsdk ignores
38 // AI_NUMERICHOST on all platforms. For numeric only internet addresses you
39 // can use SSockaddr.
40 m_hints.ai_flags &= ~AI_NUMERICHOST; // Reset flag.
41}

57// Getter for the first entry of an address info from the operating system
58// =======================================================================
59// Get address information with cached hints.
60//
61int CAddrinfo::get_first(SInaddr a_inaddr) {
62 if (m_res != nullptr)
63 throw std::runtime_error(UPnPsdk_LOGEXCEPT(
64 "MSG1189") "Object has already got address info. "
65 "Cannot get_first(inaddr) a second time.");
66
67 // If only the port is given then don't check node. Instead let
68 // ::getaddrinfo() handle special case to return "[::1]" (active), or
69 // "[::]" with AI_PASSIVE.
70 if (!(a_inaddr.node.empty() && a_inaddr.scope.empty() &&
71 !a_inaddr.service.empty())) //
72 {
73 // Doing a numeric precheck.
74 SSockaddr saObj;
75 saObj = a_inaddr;
76 if (saObj.family == AF_UNSPEC) {
77 UPnPsdk_LOGERR("MSG1190") "CAddrinfo fails with node=\""
78 << a_inaddr.node << "\", scope=\"" << a_inaddr.scope
79 << "\", service=\"" << a_inaddr.service << "\".\n";
80 return EAI_NONAME;
81 }
82 }
83
84 // Concatenate node and scope_id that needs ::getaddrinfo().
85 if (!a_inaddr.scope.empty()) {
86 // Note for later use: 'inaddr.node' is modified.
87 a_inaddr.node.reserve(a_inaddr.node.size() + 1 + a_inaddr.scope.size());
88 a_inaddr.node.append("%").append(a_inaddr.scope);
89 }
90
91 // syscall ::getaddrinfo() with prepared arguments
92 // -----------------------------------------------
93 ::addrinfo* new_res{nullptr}; // Result from UPnPsdk::getaddrinfo()
94
95 int ret = umock::netdb_h.getaddrinfo(
96 a_inaddr.node.empty() ? nullptr : a_inaddr.node.c_str(),
97 a_inaddr.service.empty() ? nullptr : a_inaddr.service.c_str(), &m_hints,
98 &new_res);
99
100 if (g_dbug) {
101 // Very helpful for debugging to see what is given to ::getaddrinfo()
102 // clang-format off
103 std::ostringstream oss;
104 oss << "syscall ::getaddrinfo("
105 << (a_inaddr.node.empty() ? "nullptr, " : "\"" + a_inaddr.node + "\", ")
106 << (a_inaddr.service.empty() ? "nullptr, " : "\"" + a_inaddr.service + "\", ")
107 << "/*hints*/" << &m_hints << ", /*res*/" << *&new_res << ") "
108 << (m_hints.ai_flags & AI_V4MAPPED ? "AI_V4MAPPED, " : "")
109 << (m_hints.ai_flags & AI_ALL ? "AI_ALL, " : "")
110 << (m_hints.ai_flags & AI_NUMERICHOST ? "AI_NUMERICHOST, " : "")
111 << (m_hints.ai_flags & AI_NUMERICSERV ? "AI_NUMERICSERV, " : "")
112 << (m_hints.ai_flags & AI_PASSIVE ? "AI_PASSIVE, " : "")
113 << (m_hints.ai_family == AF_INET6 ? "AF_INET6" :
114 (m_hints.ai_family == AF_INET ? "AF_INET" :
115 (m_hints.ai_family == AF_UNSPEC ? "AF_UNSPEC" :
116 "m_hints.ai_family=" + std::to_string(m_hints.ai_family))))
117 << ", "
118 << (m_hints.ai_socktype == SOCK_STREAM ? "SOCK_STREAM" :
119 (m_hints.ai_socktype == SOCK_DGRAM ? "SOCK_DGRAM" :
120 (m_hints.ai_socktype == SOCK_RAW ? "SOCK_RAW" :
121 "socktype=" + std::to_string(m_hints.ai_socktype))));
122 // clang-format on
123
124 if (ret == 0) {
125 SSockaddr saObj;
126 // Should be set to 'saObj.ss' to get raw data info.
127 saObj.ss = *reinterpret_cast<sockaddr_storage*>(new_res->ai_addr);
128 oss << ". Get first \"" << saObj.netaddrp()
129 << (new_res->ai_next == nullptr ? "\", no more entries."
130 : "\", more entries...");
131 UPnPsdk_LOGINFO("MSG1111") << oss.str() << "\n";
132
133 } else {
134 oss << ". Get EAI_ERROR(" << ret << ")=\"" << ::gai_strerror(ret)
135 << "\".";
136 UPnPsdk_LOGERR("MSG1111") << oss.str() << "\n";
137 }
138 } // g_dbug
139
140 // Have new valid address information with pointer 'new_res' to it?
141 // ----------------------------------------------------------------
142#if 0 // Maybe we don't need this?
143 if (ret == 0) {
144 SSockaddr saObj;
145 // Must be set to 'saObj' to get checked data.
146 saObj = *reinterpret_cast<sockaddr_storage*>(new_res->ai_addr);
147 if (saObj.family == AF_UNSPEC) {
148 UPnPsdk_LOGINFO("MSG1190") "syscall ::freeaddrinfo(" << new_res
149 << ").\n";
150 umock::netdb_h.freeaddrinfo(new_res);
151 ret = EAI_NONAME;
152 } else {
153 m_res = new_res;
154 m_res_current = new_res;
155 }
156 }
157#endif
158 if (ret == 0) {
159 m_res = new_res;
160 m_res_current = new_res;
161 }
162
163 return ret;
164}
